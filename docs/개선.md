```python
# session_manager.py 구조 제안

class SessionManager:
    def __init__(self, site_name, browser_config):
        self.site = site_name
        self.base_port = browser_config['port']
        self.process_id = None  # 브라우저 PID 추적
        self.assigned_port = None
    
    def get_available_port(self, base_port, range_limit=10):
        """포트 자동 할당 - 충돌 방지"""
        import socket
        for port in range(base_port, base_port + range_limit):
            try:
                s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
                s.bind(('127.0.0.1', port))
                s.close()
                return port
            except OSError:
                continue
        raise PortAllocationError(f"No available port between {base_port}-{base_port+range_limit}")
    
    def track_browser_process(self, port):
        """브라우저 프로세스 추적"""
        try:
            response = requests.get(f'http://127.0.0.1:{port}/json/version')
            self.process_id = response.json().get('webSocketDebuggerUrl')
            return True
        except:
            return False
```

```python
            'error': 'Invalid HTML structure'
        }
    }
    
    results = {}
    for check_name, check_spec in checks.items():
        results[check_name] = check_spec['condition']
        if not check_spec['condition']:
            logger.warning(f"Verification failed: {check_spec['error']}")
    
    return {
        'valid': all(results.values()),
        'details': results,
        'timestamp': datetime.now().isoformat()
    }

def auto_retry_download(url, max_retries=3, backoff_factor=2):
    """지수 백오프를 이용한 자동 재시도"""
    for attempt in range(max_retries):
        try:
            file_path = download_file(url)
            result = verify_download(file_path)
            if result['valid']:
                return file_path
            
            logger.info(f"Download verification failed: {result['details']}")
            
        except Exception as e:
            logger.error(f"Attempt {attempt+1} failed: {e}")
        
        # 지수 백오프: 2초, 4초, 8초
        wait_time = backoff_factor ** attempt
        logger.info(f"Retrying in {wait_time}s...")
        time.sleep(wait_time)
    
    raise DownloadVerificationError("Failed after 3 attempts")
```

```python
            return True
```

### 5. **Watchdog 시스템(Solution 3) 구현 복잡도**

**현재 계획**:
```
"Thread Monitor: 1시간 이상 'Running' 상태면 강제 종료"
```

**문제점**:
- 1시간이 모든 작업에 동일한 임계값인가?
- 강제 종료 시 진행 중인 작업 손실 위험
- 데이터베이스 Lock 문제 미처리
- Graceful shutdown vs Force kill 메커니즘 부재

**개선안**:

```python
from dataclasses import dataclass
from enum import Enum
import threading

class ThreadState(Enum):
    INITIALIZING = "initializing"
    RUNNING = "running"
    IDLE = "idle"
    HANGING = "hanging"
    DEAD = "dead"

@dataclass
class ThreadConfig:
    """스레드별 독립적 임계값"""
    name: str
    hang_threshold: int  # 초 단위
    graceful_timeout: int = 30
    max_retries: int = 3
    
    # 예시
    YOUNGRIM_DOWNLOAD = ThreadConfig("youngrim_download", hang_threshold=600)  # 10분
    ECOUNT_UPLOAD = ThreadConfig("ecount_upload", hang_threshold=300)  # 5분
    HEALTH_CHECK = ThreadConfig("health_check", hang_threshold=60)  # 1분

class Watchdog:
    def __init__(self):
        self.threads = {}
        self.hang_threshold = {
            'youngrim_download': 600,   # 10분
            'ecount_upload': 300,       # 5분
            'health_check': 60          # 1분
        }
    
    def register_thread(self, thread_name, thread_obj):
        """스레드 등록 및 모니터링 시작"""
        self.threads[thread_name] = {
            'thread': thread_obj,
            'state': ThreadState.INITIALIZING,
            'last_activity': time.time(),
            'error_count': 0
        }
    
    def detect_hang(self, thread_name) -> bool:
        """Hang 상태 감지"""
        thread_info = self.threads.get(thread_name)
        if not thread_info:
            return False
        
        elapsed = time.time() - thread_info['last_activity']
        threshold = self.hang_threshold.get(thread_name, 300)
        
        if elapsed > threshold:
            logger.warning(f"Thread '{thread_name}' may be hanging (elapsed: {elapsed}s)")
            return True
        return False
    
    def handle_hanging_thread(self, thread_name):
        """Hang 상태 스레드 처리"""
        logger.info(f"Handling hanging thread: {thread_name}")
        
        # Step 1: Graceful shutdown 시도
        if self._graceful_shutdown(thread_name, timeout=30):
            logger.info(f"Gracefully shut down {thread_name}")
            return True
        
        # Step 2: Lock 해제 시도
        if not self._release_locks(thread_name):
            logger.warning(f"Failed to release locks for {thread_name}")
        
        # Step 3: Force kill 시도
        if self._force_kill(thread_name):
            logger.info(f"Force killed {thread_name}")
            return True
        
        # Step 4: 브라우저 프로세스 강제 종료 (최후의 수단)
        if self._kill_browser_process(thread_name):
            logger.info(f"Killed browser process for {thread_name}")
            return True
        
        logger.error(f"Failed to handle hanging thread: {thread_name}")
        return False
    
    def _graceful_shutdown(self, thread_name, timeout):
        """정상 종료 신호 전송"""
        thread_info = self.threads.get(thread_name)
        if not thread_info:
            return False
        
        thread_obj = thread_info['thread']
        if hasattr(thread_obj, 'stop_event'):
            thread_obj.stop_event.set()
            thread_obj.join(timeout=timeout)
            return not thread_obj.is_alive()
        return False
    
    def _release_locks(self, thread_name) -> bool:
        """분산 Lock 해제"""
        try:
            from distributed_lock import release_all_locks
            release_all_locks(thread_name)
            logger.info(f"Released locks held by {thread_name}")
            return True
        except Exception as e:
            logger.error(f"Failed to release locks: {e}")
            return False
    
    def _force_kill(self, thread_name) -> bool:
        """Python 스레드 강제 종료 (daemon으로 변경)"""
        thread_info = self.threads.get(thread_name)
        if not thread_info:
            return False
        
        thread_obj = thread_info['thread']
        thread_obj.daemon = True
        logger.info(f"Set {thread_name} as daemon (will be killed on main exit)")
        return True
    
    def _kill_browser_process(self, thread_name) -> bool:
        """브라우저 프로세스 강제 종료"""
        try:
            if 'youngrim' in thread_name:
                os.system("pkill -f 'Avast Secure Browser'")
            elif 'ecount' in thread_name:
                os.system("pkill -f 'chrome|msedge'")
            return True
        except Exception as e:
            logger.error(f"Failed to kill browser: {e}")
            return False
    
    def monitor_continuously(self, check_interval=60):
        """연속 모니터링 (백그라운드)"""
        while True:
            for thread_name in list(self.threads.keys()):
                if self.detect_hang(thread_name):
                    self.handle_hanging_thread(thread_name)
                    # 재시작 로직
                    self._restart_thread(thread_name)
            
            time.sleep(check_interval)
    
    def _restart_thread(self, thread_name):
        """스레드 자동 재시작"""
        logger.info(f"Restarting thread: {thread_name}")
        # 구체적 재시작 로직은 각 모듈에서 구현
```

```python
from enum import Enum
from datetime import datetime, timedelta

class CircuitState(Enum):
    CLOSED = "closed"    # 정상, 요청 허용
    OPEN = "open"       # 차단, 요청 거부
    HALF_OPEN = "half_open"  # 회복 중, 제한된 요청

class CircuitBreaker:
    def __init__(self, failure_threshold=3, reset_timeout=60):
        self.failure_threshold = failure_threshold
        self.reset_timeout = reset_timeout
        self.failure_count = 0
        self.state = CircuitState.CLOSED
        self.last_failure_time = None
    
    def record_success(self):
        """성공 기록"""
        self.failure_count = 0
        self.state = CircuitState.CLOSED
    
    def record_failure(self):
        """실패 기록"""
        self.failure_count += 1
        self.last_failure_time = datetime.now()
        
        if self.failure_count >= self.failure_threshold:
            self.state = CircuitState.OPEN
            logger.warning(f"Circuit breaker OPEN after {self.failure_count} failures")
    
    def can_execute(self) -> bool:
        """요청 실행 가능 여부"""
        if self.state == CircuitState.CLOSED:
            return True
        
        if self.state == CircuitState.OPEN:
            # Reset timeout 경과 시 HALF_OPEN으로 변경
            if datetime.now() - self.last_failure_time > timedelta(seconds=self.reset_timeout):
                self.state = CircuitState.HALF_OPEN
                return True
            return False
        
        # HALF_OPEN 상태: 제한된 요청만 허용
        return True
    
    def execute(self, func, *args, **kwargs):
        """Circuit breaker를 통한 함수 실행"""
        if not self.can_execute():
            raise Exception(f"Circuit breaker is {self.state}")
        
        try:
            result = func(*args, **kwargs)
            self.record_success()
            return result
        except Exception as e:
            self.record_failure()
            raise

# 사용 예시
breaker = CircuitBreaker(failure_threshold=3, reset_timeout=60)

def login_with_circuit_breaker():
    return breaker.execute(auto_login, credentials)
```

### 2. **세션 프리페칭(Prefetching)**

```python
class SessionManager:
    def __init__(self, site_name, ttl_check_interval=3600):
        self.site_name = site_name
        self.ttl_check_interval = ttl_check_interval  # 1시간
        self.prefetch_threshold = 3600  # 1시간 전부터 갱신 준비
    
    def start_prefetch_monitor(self):
        """세션 TTL 모니터링 스레드"""
        def monitor():
            while True:
                ttl = self.get_session_ttl()
                
                if ttl < self.prefetch_threshold and ttl > 0:
                    logger.info(f"Session expiring in {ttl}s, prefetching new session...")
                    self.preemptive_relogin()
                
                time.sleep(self.ttl_check_interval)
        
        thread = threading.Thread(target=monitor, daemon=True)
        thread.start()
    
    def preemptive_relogin(self):
        """세션 만료 전 미리 갱신"""
        try:
            new_session = self.auto_login(self.credentials)
            self.save_session(new_session)
            logger.info("Prefetch completed: Session renewed before expiry")
        except Exception as e:
            logger.error(f"Prefetch failed: {e}")
```
